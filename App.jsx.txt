                                  <div className="bg-emerald-400 h-full" style={{ width: `${a.progress}%` }}></div>
                                </div>
                                <span className="font-mono text-[11px]">{a.progress}%</span>
                              </div>
                            </td>
                            <td className="py-3 text-right">
                              <button
                                onClick={() => { setSelectedAuditId(a.id); setActiveTab('inspector'); }}
                                className={`px-2.5 py-1 rounded text-xs font-semibold ${
                                  isWireframeMode ? 'border border-slate-700 bg-white' : 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/30 hover:bg-emerald-500/20'
                                }`}
                              >
                                Open Inspection
                              </button>
                            </td>
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>
                </div>

                {/* High Priority Actions Summary */}
                <div className={`p-5 rounded-2xl border flex flex-col justify-between ${isWireframeMode ? 'bg-slate-200 border-slate-400' : 'bg-slate-900/60 border-slate-800'}`}>
                  <div>
                    <h3 className="font-bold text-base mb-3 flex items-center gap-2">
                      <ShieldAlert className="w-4 h-4 text-red-400" /> High-Risk Compliance Focus
                    </h3>
                    <p className="text-xs opacity-75 mb-4">Urgent findings requiring immediate site intervention and EPA notification review.</p>

                    <div className="space-y-3">
                      {findings.filter(f => f.severity === 'High').map(f => (
                        <div key={f.id} className={`p-3 rounded-xl border text-xs space-y-1.5 ${
                          isWireframeMode ? 'border-slate-500 bg-slate-100' : 'bg-slate-950/80 border-slate-800'
                        }`}>
                          <div className="flex justify-between items-start">
                            <span className="font-bold text-red-400">{f.id}</span>
                            {getSeverityBadge(f.severity)}
                          </div>
                          <p className="font-medium text-slate-200">{f.title}</p>
                          <p className="text-[11px] opacity-60 font-mono">{f.regulation}</p>
                        </div>
                      ))}
                    </div>
                  </div>

                  <button
                    onClick={() => setActiveTab('report')}
                    className={`w-full mt-4 py-2.5 rounded-xl text-xs font-bold flex items-center justify-center gap-2 ${
                      isWireframeMode ? 'bg-slate-800 text-white' : 'bg-slate-800 hover:bg-slate-700 text-emerald-400 border border-slate-700'
                    }`}
                  >
                    <FileText className="w-4 h-4" /> Generate Full Report Draft
                  </button>
                </div>

              </div>
            </div>
          )}

          {/* ================= 2. AUDIT INSPECTOR / SITE WALK VIEW ================= */}
          {activeTab === 'inspector' && (
            <div className="space-y-6">
              
              {/* Header Toolbar & Controls */}
              <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
                <div>
                  <h2 className="text-xl font-bold tracking-tight flex items-center gap-2">
                    <Eye className="w-5 h-5 text-emerald-400" /> Site Walk & Active Audit Inspector
                  </h2>
                  <p className="text-xs opacity-75">Field data collection log for {activeAudit.facility}</p>
                </div>

                <div className="flex items-center gap-3">
                  <button
                    onClick={() => setIsNewFindingModalOpen(true)}
                    className={`px-4 py-2 rounded-lg text-xs font-bold flex items-center gap-2 cursor-pointer ${
                      isWireframeMode ? 'bg-slate-800 text-white' : 'bg-emerald-500 hover:bg-emerald-600 text-slate-950 shadow-lg shadow-emerald-500/20'
                    }`}
                  >
                    <Plus className="w-4 h-4" /> Add Field Observation
                  </button>
                </div>
              </div>

              {/* Filter and Search Bar */}
              <div className={`p-4 rounded-xl border flex flex-col md:flex-row gap-3 items-center justify-between ${
                isWireframeMode ? 'bg-slate-200 border-slate-400' : 'bg-slate-900/60 border-slate-800'
              }`}>
                <div className="relative w-full md:w-80">
                  <Search className="w-4 h-4 absolute left-3 top-2.5 opacity-50" />
                  <input
                    type="text"
                    placeholder="Search regulation, title, or notes..."
                    value={searchQuery}
                    onChange={(e) => setSearchQuery(e.target.value)}
                    className={`w-full pl-9 pr-3 py-1.5 rounded-lg text-xs focus:outline-none ${
                      isWireframeMode ? 'bg-white border border-slate-500 text-slate-900' : 'bg-slate-950 border border-slate-700 text-slate-100'
                    }`}
                  />
                </div>

                <div className="flex items-center gap-2 w-full md:w-auto overflow-x-auto pb-1 md:pb-0">
                  <Filter className="w-3.5 h-3.5 opacity-60" />
                  <span className="text-xs opacity-75 font-medium">Risk Filter:</span>
                  {['All', 'High', 'Medium', 'Low'].map((sev) => (
                    <button
                      key={sev}
                      onClick={() => setSeverityFilter(sev)}
                      className={`px-3 py-1 rounded-md text-xs font-medium cursor-pointer transition-all ${
                        severityFilter === sev
                          ? (isWireframeMode ? 'bg-slate-800 text-white font-bold' : 'bg-emerald-600 text-white')
                          : (isWireframeMode ? 'bg-slate-100 text-slate-800' : 'bg-slate-800/60 text-slate-400 hover:text-white')
                      }`}
                    >
                      {sev}
                    </button>
                  ))}
                </div>
              </div>

              {/* Findings Cards List */}
              <div className="grid grid-cols-1 gap-4">
                {filteredFindings.length === 0 ? (
                  <div className="text-center py-12 opacity-60 text-sm">
                    No audit observations found matching the selected risk filters.
                  </div>
                ) : (
                  filteredFindings.map((finding) => (
                    <div 
                      key={finding.id} 
                      className={`p-5 rounded-2xl border transition-all ${
                        isWireframeMode 
                          ? 'bg-slate-100 border-slate-400 text-slate-900' 
                          : 'bg-slate-900/80 border-slate-800 hover:border-slate-700'
                      }`}
                    >
                      <div className="flex flex-col md:flex-row md:items-center justify-between gap-3 border-b pb-3 border-slate-800/40">
                        <div className="flex items-center gap-3">
                          <span className="font-mono text-xs px-2 py-1 rounded bg-slate-800/80 text-emerald-400 border border-slate-700 font-bold">
                            {finding.id}
                          </span>
                          <h3 className="font-bold text-base">{finding.title}</h3>
                        </div>
                        <div className="flex items-center gap-2">
                          {getSeverityBadge(finding.severity)}
                          <span className={`text-xs px-2.5 py-1 rounded-full border ${
                            finding.status === 'Non-Compliant' 
                              ? 'bg-red-950/40 text-red-400 border-red-800' 
                              : 'bg-amber-950/40 text-amber-400 border-amber-800'
                          }`}>
                            {finding.status}
                          </span>
                        </div>
                      </div>

                      {/* Detail Body */}
                      <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mt-4 text-xs">
                        <div className="md:col-span-2 space-y-2">
                          <p className="opacity-90 leading-relaxed">{finding.notes}</p>

                          <div className="flex items-center gap-4 pt-2 text-[11px] opacity-75">
                            <span className="flex items-center gap-1">
                              <Building2 className="w-3.5 h-3.5 text-emerald-400" /> Location: {finding.location}
                            </span>
                            <span className="flex items-center gap-1">
                              <Camera className="w-3.5 h-3.5 text-teal-400" /> Evidence Logs: {finding.evidenceCount} Attached
                            </span>
                          </div>
                        </div>

                        {/* Regulatory Tag Box */}
                        <div className={`p-3 rounded-xl border flex flex-col justify-between space-y-2 ${
                          isWireframeMode ? 'bg-slate-200 border-slate-400' : 'bg-slate-950/60 border-slate-800/80'
                        }`}>
                          <div>
                            <span className="text-[10px] uppercase font-semibold text-emerald-400 tracking-wider">Applicable Standard</span>
                            <p className="font-bold text-xs mt-0.5">{finding.category}</p>
                            <p className="font-mono text-[11px] opacity-80">{finding.regulation}</p>
                          </div>

                          <div className="pt-2 border-t border-slate-800/60 flex justify-between items-center text-[11px]">
                            <span className="opacity-70">CAP Ref: <strong className="text-slate-200">{finding.capAssigned}</strong></span>
                            <span className="opacity-70">Due: {finding.dueDate}</span>
                          </div>
                        </div>
                      </div>
                    </div>
                  ))
                )}
              </div>
            </div>
          )}

          {/* ================= 3. REPORT GENERATOR VIEW ================= */}
          {activeTab === 'report' && (
            <div className="space-y-6">
              <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
                <div>
                  <h2 className="text-xl font-bold tracking-tight flex items-center gap-2">
                    <Sparkles className="w-5 h-5 text-amber-400" /> AI Compliance Report Draft & Generator
                  </h2>
                  <p className="text-xs opacity-75">Automated regulatory findings compilation and executive summary authoring.</p>
                </div>

                <div className="flex items-center gap-3">
                  <button
                    onClick={handleGenerateAiSummary}
                    disabled={isGeneratingAi}
                    className={`px-4 py-2 rounded-lg text-xs font-bold flex items-center gap-2 cursor-pointer transition-all ${
                      isWireframeMode
                        ? 'bg-slate-800 text-white hover:bg-slate-700'
                        : 'bg-gradient-to-r from-amber-500 to-emerald-500 hover:from-amber-600 hover:to-emerald-600 text-slate-950 shadow-lg shadow-amber-500/20'
                    }`}
                  >
                    <Sparkles className={`w-4 h-4 ${isGeneratingAi ? 'animate-spin' : ''}`} />
                    <span>{isGeneratingAi ? 'Analyzing & Writing...' : 'Generate Executive Summary with AI'}</span>
                  </button>

                  <button 
                    onClick={() => alert("Report exported as PDF (Simulated)")}
                    className={`px-3 py-2 rounded-lg text-xs font-medium flex items-center gap-2 border ${
                      isWireframeMode ? 'border-slate-500 bg-white' : 'bg-slate-800 hover:bg-slate-700 text-slate-200 border-slate-700'
                    }`}
                  >
                    <Download className="w-4 h-4" /> Export PDF
                  </button>
                </div>
              </div>

              {/* Document Editor / Live Preview Container */}
              <div className={`p-8 rounded-2xl border min-h-[500px] shadow-2xl font-serif leading-relaxed ${
                isWireframeMode ? 'bg-white text-slate-900 border-slate-400 font-mono' : 'bg-slate-900/90 border-slate-800 text-slate-200'
              }`}>
                {/* Simulated Document Header */}
                <div className="border-b pb-6 mb-6 font-sans border-slate-800">
                  <div className="flex justify-between items-start">
                    <div>
                      <span className="text-xs font-bold uppercase tracking-widest text-emerald-400">INNOVEX RISK ENGINEERING • REGULATORY REPORT</span>
                      <h1 className="text-2xl font-bold text-slate-100 font-sans mt-1">ENVIRONMENTAL COMPLIANCE AUDIT REPORT</h1>
                      <p className="text-xs opacity-70 mt-1">Facility ID: {activeAudit.facility} • Date: {activeAudit.date}</p>
                    </div>
                    <div className="text-right text-xs opacity-75 font-mono">
                      <p>Draft Ver: 1.0 AI-Pro</p>
                      <p>Status: Confidential</p>
                    </div>
                  </div>
                </div>

                {/* AI Executive Summary Block */}
                <div className="space-y-4 mb-8">
                  <h3 className="font-sans font-bold text-base text-emerald-400 uppercase tracking-wider flex items-center gap-2">
                    <FileCheck className="w-4 h-4" /> Section 1: Executive Summary & Audit Scope
                  </h3>

                  {aiGeneratedSummary ? (
                    <div className={`p-4 rounded-xl text-xs whitespace-pre-line leading-relaxed font-sans border ${
                      isWireframeMode ? 'bg-slate-100 border-slate-300' : 'bg-slate-950/80 border-slate-800 text-slate-200'
                    }`}>
                      {aiGeneratedSummary}
                    </div>
                  ) : (
                    <div className="p-8 text-center border-2 border-dashed border-slate-800 rounded-xl">
                      <Sparkles className="w-8 h-8 text-amber-400 mx-auto mb-2 opacity-60" />
                      <p className="text-xs text-slate-400 font-sans">No Executive Summary generated yet.</p>
                      <p className="text-[11px] text-slate-500 font-sans mt-1">Click "Generate Executive Summary with AI" above to automatically compile field notes into formal audit wording.</p>
                    </div>
                  )}
                </div>

                {/* Live Section 2: Detailed Findings Table */}
                <div className="space-y-4 font-sans">
                  <h3 className="font-sans font-bold text-base text-emerald-400 uppercase tracking-wider flex items-center gap-2">
                    <Layers className="w-4 h-4" /> Section 2: Summary Table of Non-Compliance Findings
                  </h3>

                  <div className="overflow-x-auto">
                    <table className="w-full text-left text-xs border border-slate-800">
                      <thead className="bg-slate-950">
                        <tr className="border-b border-slate-800 text-slate-400">
                          <th className="p-2.5">Finding Ref</th>
                          <th className="p-2.5">Category & Regulatory Standard</th>
                          <th className="p-2.5">Severity Risk</th>
                          <th className="p-2.5">CAP Tracking ID</th>
                        </tr>
                      </thead>
                      <tbody className="divide-y divide-slate-800">
                        {findings.map(f => (
                          <tr key={f.id}>
                            <td className="p-2.5 font-mono font-bold text-emerald-400">{f.id}</td>
                            <td className="p-2.5">
                              <p className="font-bold">{f.title}</p>
                              <p className="text-[11px] opacity-60">{f.regulation}</p>
                            </td>
                            <td className="p-2.5">{getSeverityBadge(f.severity)}</td>
                            <td className="p-2.5 font-mono">{f.capAssigned}</td>
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>
                </div>

              </div>
            </div>
          )}

          {/* ================= 4. TEMPLATES & REGULATIONS VIEW ================= */}
          {activeTab === 'templates' && (
            <div className="space-y-6">
              <div>
                <h2 className="text-xl font-bold tracking-tight flex items-center gap-2">
                  <BookOpen className="w-5 h-5 text-emerald-400" /> Regulatory Protocol Library & Templates
                </h2>
                <p className="text-xs opacity-75">Pre-configured EPA CFR compliance frameworks for rapid field evaluation.</p>
              </div>

              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                {REGULATORY_TEMPLATES.map(tpl => (
                  <div key={tpl.id} className={`p-5 rounded-2xl border space-y-3 ${
                    isWireframeMode ? 'bg-slate-200 border-slate-400' : 'bg-slate-900/80 border-slate-800'
                  }`}>
                    <div className="flex justify-between items-start">
                      <span className="text-xs font-mono font-bold px-2 py-0.5 rounded bg-emerald-950 text-emerald-400 border border-emerald-800">
                        {tpl.code}
                      </span>
                      <span className="text-[11px] opacity-60">{tpl.updated}</span>
                    </div>

                    <h3 className="font-bold text-base">{tpl.name}</h3>
                    <p className="text-xs opacity-75">Category: {tpl.category}</p>

                    <div className="pt-2 flex justify-between items-center text-xs">
                      <button 
                        onClick={() => { setActiveTab('inspector'); setIsNewFindingModalOpen(true); }}
                        className="text-emerald-400 font-medium hover:underline flex items-center gap-1"
                      >
                        Use Checklist Template <ChevronRight className="w-3 h-3" />
                      </button>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}

        </main>
      </div>

      {/* ================= MODAL: ADD FIELD OBSERVATION ================= */}
      {isNewFindingModalOpen && (
        <div className="fixed inset-0 bg-slate-950/80 backdrop-blur-sm z-50 flex items-center justify-center p-4">
          <div className={`w-full max-w-xl p-6 rounded-2xl border shadow-2xl space-y-4 ${
            isWireframeMode ? 'bg-slate-100 text-slate-900 border-slate-500 font-mono' : 'bg-slate-900 border-slate-800 text-slate-100'
          }`}>
            <div className="flex justify-between items-center border-b pb-3 border-slate-800">
              <h3 className="font-bold text-lg flex items-center gap-2">
                <Plus className="w-5 h-5 text-emerald-400" /> Log New Field Observation
              </h3>
              <button 
                onClick={() => setIsNewFindingModalOpen(false)}
                className="opacity-60 hover:opacity-100 cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleAddFinding} className="space-y-4 text-xs">
              <div>
                <label className="block mb-1 font-semibold">Observation Title / Non-Compliance Summary</label>
                <input
                  type="text"
                  required
                  placeholder="e.g. CEMS Analyzer Out-of-Control Period Exceeded"
                  value={newFinding.title}
                  onChange={e => setNewFinding({...newFinding, title: e.target.value})}
                  className={`w-full p-2.5 rounded-lg focus:outline-none ${
                    isWireframeMode ? 'bg-white border border-slate-500' : 'bg-slate-950 border border-slate-700'
                  }`}
                />
              </div>

              <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
                <div>
                  <label className="block mb-1 font-semibold">Regulatory Category</label>
                  <select
                    value={newFinding.category}
                    onChange={e => setNewFinding({...newFinding, category: e.target.value})}
                    className={`w-full p-2 rounded-lg ${isWireframeMode ? 'bg-white border border-slate-500' : 'bg-slate-950 border border-slate-700'}`}
                  >
                    <option>Air Quality (CEMS)</option>
                    <option>Title V Air Quality</option>
                    <option>Hazardous Waste (RCRA)</option>
                    <option>NPDES Stormwater</option>
                    <option>SPCC / Oil Containment</option>
                  </select>
                </div>

                <div>
                  <label className="block mb-1 font-semibold">Regulatory Clause Tag</label>
                  <input
                    type="text"
                    placeholder="e.g. 40 CFR Part 60.13"
                    value={newFinding.regulation}
                    onChange={e => setNewFinding({...newFinding, regulation: e.target.value})}
                    className={`w-full p-2 rounded-lg ${isWireframeMode ? 'bg-white border border-slate-500' : 'bg-slate-950 border border-slate-700'}`}
                  />
                </div>
              </div>

              <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
                <div>
                  <label className="block mb-1 font-semibold">Severity / Risk Level</label>
                  <select
                    value={newFinding.severity}
                    onChange={e => setNewFinding({...newFinding, severity: e.target.value})}
                    className={`w-full p-2 rounded-lg ${isWireframeMode ? 'bg-white border border-slate-500' : 'bg-slate-950 border border-slate-700'}`}
                  >
                    <option value="High">High Risk</option>
                    <option value="Medium">Medium Risk</option>
                    <option value="Low">Low Risk</option>
                  </select>
                </div>

                <div>
                  <label className="block mb-1 font-semibold">Location / Equipment ID</label>
                  <input
                    type="text"
                    placeholder="e.g. Stack Unit 3"
                    value={newFinding.location}
                    onChange={e => setNewFinding({...newFinding, location: e.target.value})}
                    className={`w-full p-2 rounded-lg ${isWireframeMode ? 'bg-white border border-slate-500' : 'bg-slate-950 border border-slate-700'}`}
                  />
                </div>
              </div>

              <div>
                <label className="block mb-1 font-semibold">Field Observation Notes & Evidence Description</label>
                <textarea
                  rows={3}
                  placeholder="Record exact measurements, calibration log gaps, or physical evidence noted..."
                  value={newFinding.notes}
                  onChange={e => setNewFinding({...newFinding, notes: e.target.value})}
                  className={`w-full p-2.5 rounded-lg focus:outline-none ${
                    isWireframeMode ? 'bg-white border border-slate-500' : 'bg-slate-950 border border-slate-700'
                  }`}
                />
              </div>

              <div className="pt-3 flex justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setIsNewFindingModalOpen(false)}
                  className="px-4 py-2 rounded-lg border border-slate-700 hover:bg-slate-800"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  className={`px-5 py-2 rounded-lg font-bold ${
                    isWireframeMode ? 'bg-slate-800 text-white' : 'bg-emerald-500 text-slate-950 hover:bg-emerald-600'
                  }`}
                >
                  Save Finding
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

    </div>
  );
}
